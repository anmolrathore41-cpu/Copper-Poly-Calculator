  // Cloud Sync Logic
  Future<void> fetchDataFromCloud() async {
    setState(() {
      isSyncing = true;
      syncStatus = "Syncing with Google Sheets...";
    });
    try {
      final res = await http.get(Uri.parse(apiUrl));
      if (res.statusCode == 200) {
        final data = json.decode(res.body);
        if (data['rates'] != null) {
          if (data['rates']['copper_rate'] != null) {
            copperRate = double.tryParse(data['rates']['copper_rate'].toString()) ?? copperRate;
            _copperRateCtrl.text = copperRate.toStringAsFixed(1);
          }
          if (data['rates']['film_rate'] != null) {
            filmRate = double.tryParse(data['rates']['film_rate'].toString()) ?? filmRate;
            _filmRateCtrl.text = filmRate.toStringAsFixed(1);
          }
        }
        if (data['table'] != null && (data['table'] as List).isNotEmpty) {
          wireTable = List<Map<String, dynamic>>.from(data['table'].map((item) => {
            "size": item['size'].toString(),
            "copperWt": double.tryParse(item['copperWt'].toString()) ?? 0.0,
            "filamWt": double.tryParse(item['filamWt'].toString()) ?? 0.0,
            "coilWt": double.tryParse(item['coilWt'].toString()) ?? 0.0,
            "finalOd": double.tryParse(item['finalOd'].toString()) ?? 0.0,
          }));
        }
        setState(() {
          syncStatus = "Synced with Google Sheet";
        });
      }
    } catch (e) {
      setState(() {
        syncStatus = "Offline Mode (Local)";
      });
    } finally {
      setState(() => isSyncing = false);
    }
  }

  Future<void> pushRatesToCloud() async {
    setState(() => isSyncing = true);
    try {
      await http.post(
        Uri.parse(apiUrl),
        headers: {"Content-Type": "text/plain;charset=utf-8"},
        body: json.encode({
          "action": "updateRates",
          "copperRate": copperRate,
          "filmRate": filmRate,
        }),
      );
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text("✓ Rates successfully saved to Google Sheets!")),
        );
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text("Error saving to cloud. Saved locally.")),
        );
      }
    } finally {
      setState(() => isSyncing = false);
    }
  }

  Future<void> pushTableToCloud() async {
    setState(() => isSyncing = true);
    try {
      await http.post(
        Uri.parse(apiUrl),
        headers: {"Content-Type": "text/plain;charset=utf-8"},
        body: json.encode({
          "action": "updateTable",
          "table": wireTable,
        }),
      );
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text("✓ Proportions table updated in Google Sheets!")),
        );
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text("Failed to upload table to cloud.")),
        );
      }
    } finally {
      setState(() => isSyncing = false);
    }
  }

  void showEditRowDialog(Map<String, dynamic>? existing) {
    final sizeCtrl = TextEditingController(text: existing != null ? existing['size'].toString() : "");
    final cuCtrl = TextEditingController(text: existing != null ? existing['copperWt'].toString() : "");
    final filmCtrl = TextEditingController(text: existing != null ? existing['filamWt'].toString() : "");
    final odCtrl = TextEditingController(text: existing != null ? existing['finalOd'].toString() : "");

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text(existing == null ? "Add Wire Size" : "Edit Size ${existing['size']}"),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            TextField(controller: sizeCtrl, decoration: const InputDecoration(labelText: "Size (mm)")),
            TextField(controller: cuCtrl, decoration: const InputDecoration(labelText: "Copper Wt (kg)")),
            TextField(controller: filmCtrl, decoration: const InputDecoration(labelText: "Film Wt (kg)")),
            TextField(controller: odCtrl, decoration: const InputDecoration(labelText: "Final OD (mm)")),
          ],
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text("Cancel")),
          ElevatedButton(
            child: const Text("Save"),
            onPressed: () {
              final double cu = double.tryParse(cuCtrl.text) ?? 0.0;
              final double film = double.tryParse(filmCtrl.text) ?? 0.0;
              final double coil = cu + film;

              setState(() {
                if (existing != null) {
                  existing['size'] = sizeCtrl.text;
                  existing['copperWt'] = cu;
                  existing['filamWt'] = film;
                  existing['coilWt'] = coil;
                  existing['finalOd'] = double.tryParse(odCtrl.text) ?? 0.0;
                } else {
                  wireTable.add({
                    "size": sizeCtrl.text,
                    "copperWt": cu,
                    "filamWt": film,
                    "coilWt": coil,
                    "finalOd": double.tryParse(odCtrl.text) ?? 0.0,
                  });
                }
              });
              Navigator.pop(ctx);
            },
          )
        ],
      ),
    );
  }
